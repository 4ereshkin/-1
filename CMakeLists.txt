cmake_minimum_required(VERSION 3.16)
project(InertialNavigation VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /utf-8)
else()
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra -Wpedantic")
endif()
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
add_library(navigation_model src/simulation.cpp)
target_include_directories(navigation_model PUBLIC src)
add_library(navigation_ui src/mainwindow.cpp src/mainwindow.h src/mainwindow.ui
    src/trajectoryview.cpp src/trajectoryview.h)
set_target_properties(navigation_ui PROPERTIES AUTOMOC ON AUTOUIC ON)
target_link_libraries(navigation_ui PUBLIC navigation_model Qt${QT_VERSION_MAJOR}::Widgets)
target_include_directories(navigation_ui PUBLIC src)
add_executable(inertial_navigation src/main.cpp)
target_link_libraries(inertial_navigation PRIVATE navigation_ui)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
  add_executable(test_simulation tests/test_simulation.cpp)
  target_link_libraries(test_simulation PRIVATE navigation_model)
  add_test(NAME simulation COMMAND test_simulation)
  add_executable(test_window tests/test_window.cpp)
  set_target_properties(test_window PROPERTIES AUTOMOC ON)
  target_link_libraries(test_window PRIVATE navigation_ui Qt${QT_VERSION_MAJOR}::Test)
  add_test(NAME window COMMAND test_window)
  set_tests_properties(window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
